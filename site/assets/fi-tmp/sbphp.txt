/**
 * IB Sidebar: replaces the theme "Sidebar" widget area on the front end with a designed sidebar.
 * The old widgets stay saved in Appearance > Widgets; deactivate this snippet to bring them back.
 * Ads reuse the existing Custom HTML widgets (custom_html-2, -5, -4), so the AdSense code is unchanged.
 */

if ( ! function_exists( 'ibsb_views_key' ) ) {
	function ibsb_views_key() {
		if ( function_exists( 'ibh_views_key' ) ) {
			return ibh_views_key();
		}
		$key = get_transient( 'ibh_views_key' );
		if ( false !== $key ) {
			return $key;
		}
		$key = '';
		foreach ( array( 'gamxo_views', 'post_views_count', 'views', '_post_views', 'wpb_post_views_count', 'post_views' ) as $k ) {
			$q = new WP_Query( array( 'posts_per_page' => 1, 'meta_key' => $k, 'fields' => 'ids', 'no_found_rows' => true ) );
			if ( $q->have_posts() ) { $key = $k; break; }
		}
		set_transient( 'ibh_views_key', $key, DAY_IN_SECONDS );
		return $key;
	}
}

function ibsb_ad( $n ) {
	$opt = get_option( 'widget_custom_html' );
	if ( empty( $opt[ $n ]['content'] ) ) {
		return '';
	}
	return '<div class="ibsb-ad">' . $opt[ $n ]['content'] . '</div>';
}

function ibsb_item( $id, $meta ) {
	$img = get_the_post_thumbnail_url( $id, 'thumbnail' );
	$out = '<a class="ibsb-item" href="' . esc_url( get_permalink( $id ) ) . '">';
	$out .= $img ? '<img src="' . esc_url( $img ) . '" alt="" width="64" height="64" loading="lazy">' : '<span class="ibsb-noimg"></span>';
	$out .= '<span class="ibsb-it"><span class="ibsb-t">' . esc_html( get_the_title( $id ) ) . '</span><span class="ibsb-m">' . esc_html( $meta ) . '</span></span></a>';
	return $out;
}

function ibsb_render() {
	$cur  = is_singular( 'post' ) ? get_the_ID() : 0;
	$cat  = 0;
	if ( $cur ) {
		$cat = (int) get_post_meta( $cur, 'rank_math_primary_category', true );
		if ( ! $cat ) {
			$cats = wp_get_post_categories( $cur );
			$cat  = $cats ? (int) $cats[0] : 0;
		}
	} elseif ( is_category() ) {
		$cat = (int) get_queried_object_id();
	}
	$ck   = 'ibsb_v2_' . $cur . '_' . $cat;
	$html = get_transient( $ck );
	if ( false === $html ) {
		$html = ibsb_build( $cur, $cat );
		set_transient( $ck, $html, 6 * HOUR_IN_SECONDS );
	}
	return $html;
}

function ibsb_build( $cur, $cat ) {
	$o = '';

	// 1. Search
	$o .= '<form class="ibsb-search" role="search" method="get" action="' . esc_url( home_url( '/' ) ) . '"><input type="search" name="s" placeholder="Search guides, calculators..." aria-label="Search"><button type="submit" aria-label="Search"><svg viewBox="0 0 24 24" width="18" height="18" fill="none" stroke="#0b2540" stroke-width="2.5"><circle cx="11" cy="11" r="7"/><path d="M20 20l-4-4"/></svg></button></form>';

	// 2. Ad 1
	$o .= ibsb_ad( 2 );

	// 3. Tools card
	$tools = array(
		array( 'calculators/', 'Engineering calculators', '#0077c0', '<rect x="5" y="2" width="14" height="20" rx="2"/><path d="M8 6h8M8 11h2M14 11h2M8 15h2M14 15h2M8 19h2M14 19h2"/>' ),
		array( 'unit-converters/', 'Unit converters', '#0f766e', '<path d="M7 7h11l-3-3M17 17H6l3 3"/>' ),
		array( 'tools/', 'Free engineering tools', '#6d28d9', '<path d="M14.7 6.3a4 4 0 0 0-5.4 5.4L3 18l3 3 6.3-6.3a4 4 0 0 0 5.4-5.4l-2.5 2.5-2.4-.6-.6-2.4z"/>' ),
		array( 'how-it-works/', 'How it works animations', '#f97316', '<circle cx="12" cy="12" r="9"/><path d="M10 8l6 4-6 4z"/>' ),
		array( 'downloads/', 'Free downloads', '#0b2540', '<path d="M12 3v12M7 10l5 5 5-5M5 21h14"/>' ),
	);
	$o .= '<div class="ibsb-box ibsb-tools"><div class="ibsb-h">Calculators and tools</div>';
	foreach ( $tools as $t ) {
		$o .= '<a href="' . esc_url( home_url( '/' . $t[0] ) ) . '"><span class="ibsb-ti" style="background:' . $t[2] . '"><svg viewBox="0 0 24 24" width="16" height="16" fill="none" stroke="#fff" stroke-width="2">' . $t[3] . '</svg></span>' . esc_html( $t[1] ) . '<span class="ibsb-arr">&#8250;</span></a>';
	}
	$o .= '</div>';

	// 4. Popular (in this category when known)
	$args = array( 'posts_per_page' => 5, 'post__not_in' => $cur ? array( $cur ) : array(), 'ignore_sticky_posts' => true, 'no_found_rows' => true, 'fields' => 'ids' );
	if ( $cat ) {
		$args['cat'] = $cat;
	}
	$vk = ibsb_views_key();
	if ( $vk ) {
		$args['meta_key'] = $vk;
		$args['orderby']  = 'meta_value_num';
		$args['order']    = 'DESC';
	}
	$ids = get_posts( $args );
	if ( count( $ids ) < 3 && $cat ) {
		unset( $args['cat'] );
		$ids = get_posts( $args );
		$cat = 0;
	}
	if ( $ids ) {
		$title = $cat ? 'Popular in ' . get_cat_name( $cat ) : 'Most viewed guides';
		$o    .= '<div class="ibsb-box"><div class="ibsb-h">' . esc_html( $title ) . '</div><div class="ibsb-rank">';
		$i     = 0;
		foreach ( $ids as $id ) {
			$i++;
			$v  = $vk ? (int) get_post_meta( $id, $vk, true ) : 0;
			$m  = $v ? ( $v >= 1000 ? round( $v / 1000, 1 ) . 'k views' : $v . ' views' ) : get_the_date( 'M j, Y', $id );
			$o .= '<a class="ibsb-r" href="' . esc_url( get_permalink( $id ) ) . '"><span class="ibsb-n">' . $i . '</span><span class="ibsb-it"><span class="ibsb-t">' . esc_html( get_the_title( $id ) ) . '</span><span class="ibsb-m">' . esc_html( $m ) . '</span></span></a>';
		}
		$o .= '</div></div>';
	}

	// 5. Ad 2
	$o .= ibsb_ad( 5 );

	// 6. Topics chips
	$cols  = array( '#0077c0', '#0f766e', '#6d28d9', '#f97316', '#0b2540', '#be185d', '#15803d', '#b45309' );
	$terms = get_terms( array( 'taxonomy' => 'category', 'orderby' => 'count', 'order' => 'DESC', 'number' => 14, 'hide_empty' => true, 'exclude' => array( 1 ) ) );
	if ( $terms && ! is_wp_error( $terms ) ) {
		$o .= '<div class="ibsb-box"><div class="ibsb-h">Explore topics</div><div class="ibsb-chips">';
		foreach ( $terms as $k => $t ) {
			$c  = $cols[ $k % count( $cols ) ];
			$o .= '<a href="' . esc_url( get_term_link( $t ) ) . '" style="--c:' . $c . '">' . esc_html( $t->name ) . '<b>' . (int) $t->count . '</b></a>';
		}
		$o .= '</div></div>';
	}

	// 7. Latest: first one as a full width 16:9 card (featured image stays readable), the rest as text rows
	$latest = get_posts( array( 'posts_per_page' => 4, 'post__not_in' => $cur ? array( $cur ) : array(), 'ignore_sticky_posts' => true, 'no_found_rows' => true, 'fields' => 'ids' ) );
	if ( $latest ) {
		$o .= '<div class="ibsb-box"><div class="ibsb-h">Latest articles</div>';
		foreach ( $latest as $k => $id ) {
			$cats = get_the_category( $id );
			$cn   = $cats ? $cats[0]->name : '';
			if ( 0 === $k ) {
				$img = get_the_post_thumbnail_url( $id, 'medium_large' );
				$o  .= '<a class="ibsb-lead" href="' . esc_url( get_permalink( $id ) ) . '">' . ( $img ? '<img src="' . esc_url( $img ) . '" alt="' . esc_attr( get_the_title( $id ) ) . '" width="768" height="432" loading="lazy">' : '' ) . '<span class="ibsb-t">' . esc_html( get_the_title( $id ) ) . '</span><span class="ibsb-m">' . esc_html( $cn ? $cn . ' · ' . get_the_date( 'M j, Y', $id ) : get_the_date( 'M j, Y', $id ) ) . '</span></a>';
			} else {
				$o .= '<a class="ibsb-line" href="' . esc_url( get_permalink( $id ) ) . '">' . ( $cn ? '<span class="ibsb-cat">' . esc_html( $cn ) . '</span>' : '' ) . '<span class="ibsb-t">' . esc_html( get_the_title( $id ) ) . '</span><span class="ibsb-m">' . esc_html( get_the_date( 'M j, Y', $id ) ) . '</span></a>';
			}
		}
		$o .= '</div>';
	}

	// 8. Community
	$o .= '<div class="ibsb-comm"><div class="ibsb-ch">Get new guides first</div><p>New guides and calculators, the day they go live.</p>'
		. '<a class="wa" href="https://whatsapp.com/channel/0029Vare99K3mFY3QVjKF43r" target="_blank" rel="noopener">WhatsApp channel</a>'
		. '<a class="tg" href="https://t.me/Instrumentationblogs" target="_blank" rel="noopener">Telegram channel</a>'
		. '<a class="sp" href="' . esc_url( home_url( '/donate-us-instrumentationblog/' ) ) . '">&#9829; Support free learning</a></div>';

	// 9. Ad 3
	$o .= ibsb_ad( 4 );

	return $o;
}

// Front end only: keep the original language widget (gtranslate) and drop the other old widgets.
add_filter( 'sidebars_widgets', function ( $w ) {
	if ( is_admin() || wp_doing_ajax() || ( defined( 'REST_REQUEST' ) && REST_REQUEST ) || is_customize_preview() || ! did_action( 'template_redirect' ) ) {
		return $w;
	}
	if ( ! empty( $w['sidebar'] ) && is_array( $w['sidebar'] ) ) {
		$w['sidebar'] = array_values( array_filter( $w['sidebar'], function ( $id ) {
			return 0 === strpos( $id, 'gtranslate' );
		} ) );
	}
	return $w;
} );

// Designed sidebar goes right after the language widget.
add_action( 'dynamic_sidebar_after', function ( $index ) {
	if ( 'sidebar' !== $index || is_admin() ) {
		return;
	}
	echo '<div class="ibsb">' . ibsb_render() . '</div>'; // phpcs:ignore
}, 99 );

// Clear cached sidebars when a post is published or updated.
add_action( 'save_post_post', function () {
	global $wpdb;
	$wpdb->query( "DELETE FROM {$wpdb->options} WHERE option_name LIKE '\_transient\_ibsb\_v2\_%' OR option_name LIKE '\_transient\_timeout\_ibsb\_v2\_%'" ); // phpcs:ignore
} );
